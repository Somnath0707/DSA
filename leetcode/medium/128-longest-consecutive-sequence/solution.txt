              if(indexMap.containsKey(nums[i]-1))
                dsu.union(indexMap.get(nums[i]) , indexMap.get(nums[i]-1));
        }

        int max = 0 ; 
        Map<Integer,Integer> map = new HashMap<>(); 
        for(int i = 0 ; i < n ; i++){
            if(set.contains(nums[i])) continue; 
            int parent = dsu.find(indexMap.get(nums[i]));
            map.put(parent , map.getOrDefault(parent , 0)+1);
            max = Math.max(max , map.get(parent));
            set.add(nums[i]);
        }

        return max; 
    }
