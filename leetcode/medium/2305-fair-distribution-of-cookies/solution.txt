class Solution {
    public int f(int i, int total, int[] map, int[] nums, int k) {
        if(i == nums.length){
            return total;
        }

        int ans = Integer.MAX_VALUE;

        for(int ind = 0; ind < k; ind++){
            map[ind] += nums[i];

            int newTotal = Math.max(total, map[ind]);

            if(newTotal < ans){
                ans = Math.min(ans, f(i + 1, newTotal, map, nums, k));
            }

            map[ind] -= nums[i];

            if(map[ind] == 0)
                break;
        }

        return ans;
    }
