 , int total , Map<Integer ,Integer> map , int []nums , int k ){
        if(i == nums.length){// all taken return 
            return total; 
        }

        int ans = Integer.MAX_VALUE; 

        for(int ind = 1 ; ind <= k ; ind++){
            // if taken skip 

            map.put(ind , map.getOrDefault(ind , 0) + nums[i] );
            // maks it ; 
            int newTotal = Math.max(total , map.get(ind)); 
            ans = Math.min(ans , f(i+1, newTotal , map , nums , k));
            
            map.put(ind , map.get(ind)-nums[i] );
        }

        return ans; 
    }

    public int distributeCookies(int[] cookies, int k) {
        return f(0 ,0 ,  new HashMap<>() , cookies , k);
    }
