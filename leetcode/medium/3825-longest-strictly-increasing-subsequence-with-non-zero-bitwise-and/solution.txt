class Solution {
    public int longestSubsequence(int[] nums) {
        int[] tails = new int[nums.length];
        int maxLen = 0;
        
        for (int b = 0; b < 31; b++) {
            int len = 0;
            for (int x : nums) {
                if ((x & (1 << b)) == 0) continue; 
                
                int i = 0, j = len;
                while (i < j) {
                    int m = (i + j) / 2;
                    if (tails[m] < x)
                        i = m + 1;
                    else
                        j = m;
                }
                tails[i] = x;
                if (i == len)
                    len++;
            }
            maxLen = Math.max(maxLen, len);
        }
        return maxLen;
    }
}
