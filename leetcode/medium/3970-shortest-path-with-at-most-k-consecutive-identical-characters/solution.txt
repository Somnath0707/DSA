 newCost = cost + nextCost ; 
                if(newCost < dist[nextNode][newRun]){
                    dist[nextNode][newRun] = newCost; 
                    pq.offer(new int[]{newCost , nextNode , newRun});
                }
                
                
            }
        }

        int ans = Integer.MAX_VALUE; 

        for(int i = 1 ; i <= k ; i++){
            ans = Math.min(ans , dist[n-1][i]);
        }
        if(ans == Integer.MAX_VALUE) return -1 ; 
        return ans; 
    }
