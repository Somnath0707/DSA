            map.put(str1 , str2); 
        }

        StringBuilder str = new StringBuilder(); 
        int start = -1; 
        for(int i = 0 ; i < n ; i++){
            char curr = s.charAt(i); 
            if(curr == '('){
                start = i ;
                isOn = true; 
            }
            if(curr == ')'){
                String temp = s.substring(start + 1 , i); 
                String val = (map.containsKey(temp)) ? map.get(temp) : "?";
                str.append(val);
                isOn = false; 
                start = -1; 
            }
            else if(!isOn){
                str.append(curr); 
            }
        }

        return str.toString();
