        ans++; 
                    }else{
                        if(evenPair > 0){
                            evenPair--; 
                            oddPair++; 
                            ans++; 
                        }
                        else{
                            break; 
                        }
                    }
                }
                else break ;

            }
        }
        return ans; 
    }
}
