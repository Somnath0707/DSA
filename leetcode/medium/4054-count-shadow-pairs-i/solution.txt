class Solution {
    int bs(int left, int right, int nums[], int val) {
        int ans = -1; 
        while(left <= right){
            int mid = left + (right - left) / 2; 
            if(nums[mid] < val){
                ans = mid; 
                left = mid + 1; 
            } else {
                right = mid - 1; 
            }
        }
        
        return ans + 1; 
    }
