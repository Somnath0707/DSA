      // 1 2 3 4 
        long ans = 0; 
        int n = nums.length; 
        int monoStack[] = new int[n]; 
        int ind = -1; 

        for(int i = 0 ; i < n ; i++){
            int curr = nums[i]; 

            while(ind >= 0 && monoStack[ind] > curr){
                int last = bs(0 , ind , monoStack , monoStack[ind]); 
                ans += last; 
                monoStack[ind] = 0;
                ind--; 
            }

            ind++;
            monoStack[ind] = curr; 
        }

        while(ind >= 0){
            int last = bs(0 , ind , monoStack , monoStack[ind]);
            ans += last; 
            monoStack[ind] = 0;
            ind--; 
        }

        return ans; 
    }
