
    public int scoreOfParentheses(String s) {
        int total = 0;
        Stack<Integer> st = new Stack<>();
        st.push(0);

        for (char ch : s.toCharArray()) {
            if (ch == '(') {
                st.push(0);
            } else {
                int inside = st.pop();

                if (inside == 0)
                    st.push(st.pop() + 1);
                else
                    st.push(st.pop() + 2 * inside);
            }
        }

        return st.pop();
    }
