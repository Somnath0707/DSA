class Solution {
    public int countIntersectingIntervals(int[][] nums) {
        int ans = 0;

        for (int i = 0; i < nums.length - 1; i++) {
            int start = nums[i][0];
            int end = nums[i][1];

            for (int j = i + 1; j < nums.length; j++) {
                int currStart = nums[j][0];
                int currEnd = nums[j][1];

                if (currStart <= end && start <= currEnd) {
                    ans++;
                }
            }
        }

        return ans;
    }
}
