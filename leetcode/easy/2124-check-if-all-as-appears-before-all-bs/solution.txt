class Solution {
    public boolean checkString(String s) {
        boolean seen = false; 
        for(int i = 0 ; i < s.length() ; i++){
            if(seen == true && s.charAt(i) == 'a'){
                return false; 
            }
            if(s.charAt(i) == 'b') seen = true; 
        }
        return true; 
    }
}
