class Solution {
    public boolean checkStr(String str , int ind){
        int ans = 0 ; 
        for(int i = 0 ; i < str.length() ; i++){
            ans += str.charAt(i) -'0';
        }
        if( ans == ind) return true; 
        return false; 
    }
    public int smallestIndex(int[] nums) {
        for(int i = 0 ; i < nums.length ; i++){
            String str = String.valueOf(nums[i]);

            if(checkStr(str , i)) return i ; 
        }
        return -1; 
    }
}
