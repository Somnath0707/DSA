ic String[] createGrid(int m, int n) {
        String ans[] = new String[m];
        char [][] grid = new char[m][n]; 

        for(int i = 0 ; i < m ; i++){
            Arrays.fill(grid[i] , '#');
            
        }

        for(int j = 0 ; j  < n ; j++){
            grid[0][j] = '.';
        }

        for(int i = 0 ; i < m ;i++){
            grid[i][n-1] = '.';
        }

        for(int i = 0 ; i < m ; i++){
            ans[i] = new String(grid[i]);
        }

        return ans;
    }
}
