class Solution {
    int n;
    Integer dp[][];

    public int gcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }

    public int f(int mask, int currOp, int nums[]) {
        if (mask == (1 << n) - 1) {
            return 0;
        }

        if (dp[mask][currOp] != null) return dp[mask][currOp];

        int ans = 0;

        for (int i = 0; i < n; i++) {
            if ((mask & (1 << i)) != 0) continue;

            for (int j = i + 1; j < n; j++) {
                if ((mask & (1 << j)) != 0) continue;

                int newMask = mask | (1 << i) | (1 << j);

                ans = Math.max(ans,
                    currOp * gcd(nums[i], nums[j])
                    + f(newMask, currOp + 1, nums));
            }
        }

        return dp[mask][currOp] = ans;
    }
