b;
            a = temp;
        }
        return a;
    }

    public int f(int mask, int currOp, int nums[]) {
        if (mask == (1 << n) - 1) {
            return 0;
        }

        if (dp[mask] != null) return dp[mask];

        int ans = 0;

        for (int i = 0; i < n; i++) {
            if ((mask & (1 << i)) != 0) continue;

            for (int j = i + 1; j < n; j++) {
                if ((mask & (1 << j)) != 0) continue;

                int newMask = mask | (1 << i) | (1 << j);

                ans = Math.max(ans,
                    currOp * gcd(nums[i], nums[j])
                    + f(newMask, currOp + 1, nums));
            }
        }

        return dp[mask] = ans;
    }

    public int maxScore(int[] nums) {
        n = nums.length;
        dp = new Integer[1 << n];
        return f(0, 1, nums);
    }
