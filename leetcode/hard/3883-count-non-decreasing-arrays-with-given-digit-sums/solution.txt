    public int f(int i , int sum , int [] nums){
        if(i == nums.length) return 1 ; 

        if( sum > 5000 ) return 0 ; 
        if(dp[i][sum] != -1) return dp[i][sum]; 
        int ans = 0 ; 

        if(check(sum , nums[i])){
            ans = (ans + f(i+1 , sum , nums) )% MOD;
        }

        ans = (ans + f(i , sum+1 , nums)) %  MOD; 


        return dp[i][sum] = ans; 
    }
    public int countArrays(int[] digitSum) {
    //   1,50,00,000
        dp = new int[digitSum.length][5001];
        for(int i = 0 ; i < digitSum.length ; i++){
            Arrays.fill(dp[i] , -1);
        }
        return f(0 , 0 , digitSum); 
    }
