nt f(int i, int j, int steps, int[][] mat) {
        if (check(mat))
            return steps;
        if (i == mat.length)
            return Integer.MAX_VALUE;

        int ans = Integer.MAX_VALUE;

        for (int ind = i; ind < mat.length; ind++) {
            for (int k = j; k < mat[0].length; k++) {
                int nextI = ind;
                int nextJ = k + 1;

                if (nextJ == mat[0].length) {
                    nextI++;
                    nextJ = 0;
                }

                int temp[][] = change(ind, k, mat);
                int res = f(nextI, nextJ, steps + 1, temp);
                ans = Math.min(ans, res);
            }
        }

        return ans;
    }

    public int minFlips(int[][] mat) {
        int ans = f(0, 0, 0, mat);
        return ans == Integer.MAX_VALUE ? -1 : ans;
    }
