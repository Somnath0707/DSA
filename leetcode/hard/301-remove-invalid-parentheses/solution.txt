class Solution {
    Set<String> list; 
    int min; 
    
    Integer dp[][][];
    public void f(int i, StringBuilder str, int total, int remove, String s){
        if(total < 0) return;
        if(remove > min) return ;  
        
        if(i == s.length()){
            if(remove == min && total == 0){
                list.add(str.toString()); 
            }
            return;
        }

        f(i+1, str, total, remove+1, s); 

        char curr = s.charAt(i); 

        int newTotal = total; 
        if(curr == '('){
            newTotal = total+1; 
        }else if(curr == ')'){
            newTotal = total-1; 
        }


        str.append(curr); 
        f(i+1, str, newTotal, remove, s); 
        str.setLength(str.length()-1);
    }
