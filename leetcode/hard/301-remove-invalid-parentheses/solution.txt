!= null) return dp[i][total][remove]; 

        int skip = dfs(i+1, total, remove+1, s); 

        char curr = s.charAt(i); 

        int newTotal = total; 
        if(curr == '('){
            newTotal = total+1; 
        }else if(curr == ')'){
            newTotal = total-1; 
        }

        int take = dfs(i+1, newTotal, remove, s); 

        return dp[i][total][remove] = Math.min(take, skip);
    }
    
    public List<String> removeInvalidParentheses(String s) {
        int n = s.length() ; 
        dp = new Integer[n][n][n]; 
        min = dfs(0, 0, 0, s); 
        list = new  HashSet<>(); 
        List<String> ans = new ArrayList<>(); 

        f(0, new StringBuilder(), 0, 0, s);
        for(String str : list){
            ans.add(str); 
        }

        return ans; 
    }
