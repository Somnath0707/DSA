s);
        Arrays.sort(target);

        int n = nums.length;

        long pos = 0;
        long neg = 0;
        long count = 0;

        // odd numbers
        for (int i = 0, j = 0; i < n; i++) {
            if (nums[i] % 2 != 0) {
                while (target[j] % 2 == 0) j++;

                long diff = nums[i] - target[j];

                if (diff > 0)
                    pos += diff;
                else
                    neg += -diff;

                j++;
            }
        }

        for (int i = 0, j = 0; i < n; i++) {
            if (nums[i] % 2 == 0) {
                while (target[j] % 2 != 0) j++;

                long diff = nums[i] - target[j];

                if (diff > 0)
                    pos += diff;
                else
                    neg += -diff;

                j++;
            }
        }

        count = pos / 2;

        return count;
    }
