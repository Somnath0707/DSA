nteger>> list = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            list.add(new ArrayList<>());
        }

        for (int e[] : edges) {
            int u = e[0];
            int v = e[1];

            list.get(u).add(v);
            list.get(v).add(u);
        }

        int ans = Integer.MAX_VALUE;

        for (int i = 0; i < n; i++) {
            ans = Math.min(ans, bfs(i, list, n));
        }

        if (ans == Integer.MAX_VALUE) return -1;

        return ans;
    }
