class Solution {
    String dp[][];
    int n;

    public int overlap(String a, String b) {
        int len = Math.min(a.length(), b.length());

        for (int k = len; k >= 0; k--) {
            if (a.substring(a.length() - k).equals(b.substring(0, k))) {
                return k;
            }
        }

        return 0;
    }
