    skip = f(i + 1, taken, currentBeauty, max, k);

        return (int) ((take + skip) % MOD);
    }

    public int countKSubsequencesWithMaxBeauty(String s, int k) {

        for (char ch : s.toCharArray()) {
            map.put(ch, map.getOrDefault(ch, 0) + 1);
        }

        if (map.size() < k) return 0;

        PriorityQueue<Integer> pq = new PriorityQueue<>();

        for (int num : map.values()) {
            pq.offer(num);

            if (pq.size() > k)
                pq.poll();
        }

        int max = 0;

        while (!pq.isEmpty()) {
            max += pq.poll();
        }

        // Start recursion: checking alphabet at index 0, taken 0, currentBeauty 0
        return f(0, 0, 0, max, k);
    }
