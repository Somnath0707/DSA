       return dp[i][j][k] = (int)((take1 + take2) % MOD);
    }

    public int interleaveCharacters(String word1, String word2, String target) {

        int n = word1.length();
        int n2 = word2.length();
        int n3 = target.length();

        dp = new int[n + 1][n2 + 1][n3 + 1];

        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= n2; j++) {
                Arrays.fill(dp[i][j], -1);
            }
        }

        return f(0, 0, 0, word1, word2, target);
    }
