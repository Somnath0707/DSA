r][nc][totalOpen][newClose] = true;
                        q.offer(new int[]{nr, nc, totalOpen, newClose});
                    }

                } else {
                    int newOpen = totalOpen + 1;

                    if (!vis[nr][nc][newOpen][totalClose]) {
                        vis[nr][nc][newOpen][totalClose] = true;
                        q.offer(new int[]{nr, nc, newOpen, totalClose});
                    }
                }
            }
        }

        return false;
    }
