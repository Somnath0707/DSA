   boolean down = false;
        boolean right = false;

        if (i + 1 < n) {
            down = down || f(i + 1, j, net);
        }

        if (j + 1 < m) {
            right = right || f(i, j + 1, net);
        }

        return dp[i][j][net] = down || right;
    }

    public boolean hasValidPath(char[][] grid) {
        this.grid = grid;
        n = grid.length;
        m = grid[0].length;

        if (grid[0][0] == ')' || grid[n - 1][m - 1] == '(') {
            return false;
        }

        dp = new Boolean[n][m][n + m + 1];

        return f(0, 0, 0);
    }
