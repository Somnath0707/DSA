class Solution {
    Boolean dp[][][]; 
    public boolean  f(int i , int j , int net , char grid[][]){

        if(i == grid.length-1 && j == grid[0].length-1){

            if(net == 1 && grid[i][j] == ')') return true;  
            return false; 
        }

        if(dp[i][j][net] != null) return dp[i][j][net]; 

        

        boolean down = false; 
        boolean right = false; 
      
        int newNet = (grid[i][j] == ')') ? net-1 : net+1; 
        if(i+1 < grid.length && newNet >= 0 )
        down = down || f(i+1 , j , newNet , grid); 

        if(j+1 < grid[0].length && newNet >= 0)
        right = right || f(i , j+1 , newNet , grid); 


        return dp[i][j][net] = down || right; 
        

        
    }
