    public int f(int i, int nums[][]) {
        if (i == nums.length) return 0;
        if(dp[i] != null) return dp[i]; 
        int take = 0;
        int skip = 0;

        int start = nums[i][0];
        int end = nums[i][1];

        int findNext = binarySearch(nums, i + 1, end);

        take = nums[i][2] + f(findNext, nums);

        skip = f(i + 1, nums);

        return dp[i] = Math.max(take, skip);
    }
