class Solution {
    Long dp[][]; 

    public int binarySearch(int nums[][], int start, int end) {
        int left = start;
        int right = nums.length - 1;
        int ans = nums.length;

        while (left <= right) {
            int mid = left + (right - left) / 2;

            if (nums[mid][0] >= end) {
                ans = mid;
                right = mid - 1;
            } else {
                left = mid + 1;
            }
        }

        return ans;
    }
