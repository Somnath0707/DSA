 ; i < n ; i++){
            temp[i][0] = plantTime[i];
            temp[i][1] = growTime[i]; 
        }

        Arrays.sort(temp , (a,b)->{
            if(a[1] != b[1]) return (b[1]-a[1]);
            return b[0]-a[0]; 
            }
        );

        int start = 0 ; 
        int last = 0 ; 
        for(int i = 0 ; i < n ; i++){
            int currTime = temp[i][0]; 
            int currGrow = temp[i][1]; 

            start += currTime; 
            last = Math.max(last , start + currGrow); 
        }
        return last; 
    }
