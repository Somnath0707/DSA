            map.putIfAbsent(parent , new TreeSet<>());

        }

        int ans[][] = new int[map.size()][2];
        int ind = 0 ; 
        for(Map.Entry<Integer,TreeSet<Integer>> entry : map.entrySet()){

            ans[ind][0] = val.first();

        }
